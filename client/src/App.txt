import React, { useState, useEffect } from 'react';
import logo from './assets/logo.png';

export default function App() {
  const [currentView, setCurrentView] = useState('login');
  
  // Login fields
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [loggedInUser, setLoggedInUser] = useState(null);
  const [loginError, setLoginError] = useState('');

  // Navbar user dropdown state
  const [isUserMenuOpen, setIsUserMenuOpen] = useState(false);

  // Data states
  const [records, setRecords] = useState([]);
  
  // Filter & Search states
  const [selectedModule, setSelectedModule] = useState('ALL');
  const [selectedAssessment, setSelectedAssessment] = useState('ALL');
  const [searchQuery, setSearchQuery] = useState('');

  // Notification Banner State (Accessibility / Usability)
  const [notification, setNotification] = useState({ message: '', type: '' });

  // Modal State
  const [activeModalRecord, setActiveModalRecord] = useState(null);
  const [isEditing, setIsEditing] = useState(false);
  const [editFormData, setEditFormData] = useState({});

  // Form state for creating a new record
  const [newRecord, setNewRecord] = useState({
    student_name: '',
    student_id: '',
    module_code: 'CS101 - Algorithms',
    assessment: 'Final Exam',
    score: '',
    status: 'Approved',
    review_reason: '',
  });

  useEffect(() => {
    if (currentView === 'dashboard') {
      fetchRecords();
    }
  }, [currentView]);

  // Accessibility: Listen for 'Esc' key to close modal or user menu
  useEffect(() => {
    const handleKeyDown = (e) => {
      if (e.key === 'Escape') {
        if (activeModalRecord) {
          setActiveModalRecord(null);
          setIsEditing(false);
        }
        if (isUserMenuOpen) {
          setIsUserMenuOpen(false);
        }
      }
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [activeModalRecord, isUserMenuOpen]);

  const showNotification = (message, type = 'success') => {
    setNotification({ message, type });
    setTimeout(() => setNotification({ message: '', type: '' }), 4000);
  };

  const fetchRecords = async () => {
    try {
      const res = await fetch('http://localhost:3000/api/records');
      const data = await res.json();
      setRecords(data);
    } catch (err) {
      console.error('Failed to fetch records', err);
    }
  };

  const handleLogin = async (e) => {
    e.preventDefault();
    setLoginError('');
    try {
      const res = await fetch('http://localhost:3000/api/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, password })
      });
      const data = await res.json();
      
      if (res.ok) {
        setLoggedInUser(data.user);
        setCurrentView('dashboard');
        showNotification(`Welcome back, Prof. ${data.user.user_name}.`);
      } else {
        setLoginError(data.error || 'Invalid credentials');
      }
    } catch (err) {
      setLoginError('Server connection failed');
    }
  };

  const handleAddRecord = async (e) => {
    e.preventDefault();
    try {
      const payload = {
        ...newRecord,
        score: Number(newRecord.score),
        last_modifiedby: loggedInUser ? loggedInUser.user_name : 'Admin'
      };

      const res = await fetch('http://localhost:3000/api/records', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });

      if (res.ok) {
        fetchRecords();
        setCurrentView('dashboard');
        setNewRecord({
          student_name: '',
          student_id: '',
          module_code: 'CS101 - Algorithms',
          assessment: 'Final Exam',
          score: '',
          status: 'Approved',
          review_reason: '',
        });
        showNotification('New exam record successfully created.');
      }
    } catch (err) {
      console.error('Failed to create record', err);
    }
  };

  const handleUpdateRecord = async (e) => {
    e.preventDefault();
    try {
      const payload = {
        ...editFormData,
        score: Number(editFormData.score),
        last_modifiedby: loggedInUser ? loggedInUser.user_name : 'Admin'
      };

      const res = await fetch(`http://localhost:3000/api/records/${editFormData.id}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });

      if (res.ok) {
        fetchRecords();
        setIsEditing(false);
        setActiveModalRecord(null);
        showNotification(`Record #${editFormData.id} updated successfully.`);
      }
    } catch (err) {
      console.error('Failed to update record', err);
    }
  };

  const handleDelete = async (id) => {
    if (!window.confirm('Are you sure you want to delete this record?')) return;
    try {
      const res = await fetch(`http://localhost:3000/api/records/${id}`, {
        method: 'DELETE'
      });
      if (res.ok) {
        setActiveModalRecord(null);
        fetchRecords();
        showNotification(`Record #${id} deleted successfully.`, 'error');
      }
    } catch (err) {
      console.error('Failed to delete record', err);
    }
  };

  const openModal = (record) => {
    setActiveModalRecord(record);
    setEditFormData(record);
    setIsEditing(false);
  };

  // Filter & Search logic (Module + Assessment Type + Student Search)
  const filteredRecords = records.filter(record => {
    const matchesModule = selectedModule === 'ALL' || record.module_code === selectedModule;
    const matchesAssessment = selectedAssessment === 'ALL' || record.assessment === selectedAssessment;
    const matchesSearch = searchQuery === '' || 
      record.student_name.toLowerCase().includes(searchQuery.toLowerCase()) || 
      record.student_id.toLowerCase().includes(searchQuery.toLowerCase());

    return matchesModule && matchesAssessment && matchesSearch;
  });

  // 1. LOGIN VIEW
  if (currentView === 'login') {
    return (
      <main className="app-container login-container">
        <section className="login-card">
          <header>
            <img src={logo} alt="OmniKey Logo" className="logoh" />
            <p className="portal-subtitle">Secure University Exam Records Desk</p>
          </header>
          {loginError && <div className="alert error" role="alert">{loginError}</div>}
          <form onSubmit={handleLogin} className="login-form">
            <div className="form-group">
              <label htmlFor="email">Email Address</label>
              <input 
                id="email"
                type="email" 
                value={email} 
                onChange={(e) => setEmail(e.target.value)} 
                placeholder="eyad@university.edu"
                required 
              />
            </div>
            <div className="form-group">
              <label htmlFor="password">Password</label>
              <input 
                id="password"
                type="password" 
                value={password} 
                onChange={(e) => setPassword(e.target.value)} 
                placeholder="12345678"
                required 
              />
            </div>
            <button type="submit" className="btn primary-btn">Sign In</button>
          </form>
        </section>
      </main>
    );
  }

  // 2. DATA-ENTRY FORM VIEW
  if (currentView === 'form') {
    return (
      <main className="app-container form-container">
        <header className="view-header">
          <h1>Add New Exam Record</h1>
          <button onClick={() => setCurrentView('dashboard')} className="btn secondary-btn">Back to Dashboard</button>
        </header>

        <form onSubmit={handleAddRecord} className="record-form">
          <div className="form-group">
            <label htmlFor="student_name">Student Full Name</label>
            <input 
              id="student_name"
              type="text" 
              value={newRecord.student_name} 
              onChange={(e) => setNewRecord({...newRecord, student_name: e.target.value})} 
              required 
            />
          </div>

          <div className="form-group">
            <label htmlFor="student_id">Student ID</label>
            <input 
              id="student_id"
              type="text" 
              value={newRecord.student_id} 
              onChange={(e) => setNewRecord({...newRecord, student_id: e.target.value})} 
              required 
            />
          </div>

          <div className="form-group">
            <label htmlFor="module_code">Module Code</label>
            <select 
              id="module_code"
              value={newRecord.module_code} 
              onChange={(e) => setNewRecord({...newRecord, module_code: e.target.value})}
            >
              <option value="CS101 - Algorithms">CS101 - Algorithms</option>
              <option value="AI402 - Neural Networks">AI402 - Neural Networks</option>
              <option value="SE305 - Software Engineering">SE305 - Software Engineering</option>
            </select>
          </div>

          <div className="form-group">
            <label htmlFor="assessment">Assessment Type</label>
            <select 
              id="assessment"
              value={newRecord.assessment} 
              onChange={(e) => setNewRecord({...newRecord, assessment: e.target.value})}
            >
              <option value="Final Exam">Final Exam</option>
              <option value="Midterm">Midterm</option>
              <option value="Coursework">Coursework</option>
            </select>
          </div>

          <div className="form-group">
            <label htmlFor="score">Score (0-100)</label>
            <input 
              id="score"
              type="number" 
              min="0"
              max="100"
              value={newRecord.score} 
              onChange={(e) => setNewRecord({...newRecord, score: e.target.value})} 
              required 
            />
          </div>

          <div className="form-group">
            <label htmlFor="status">Grade Status</label>
            <select 
              id="status"
              value={newRecord.status} 
              onChange={(e) => setNewRecord({...newRecord, status: e.target.value})}
            >
              <option value="Approved">Approved</option>
              <option value="Pending Review">Pending Review</option>
              <option value="Grade Appeal">Grade Appeal</option>
            </select>
          </div>

          {(newRecord.status === 'Pending Review' || newRecord.status === 'Grade Appeal') && (
            <div className="form-group dynamic-field">
              <label htmlFor="review_reason">Review / Appeal Reason</label>
              <input 
                id="review_reason"
                type="text" 
                value={newRecord.review_reason} 
                onChange={(e) => setNewRecord({...newRecord, review_reason: e.target.value})} 
                required 
              />
            </div>
          )}

          <button type="submit" className="btn primary-btn">Save Record to Database</button>
        </form>
      </main>
    );
  }

  // 3. DASHBOARD VIEW
  return (
    <div className="dashboard-wrapper">
      {/* TOP NAVIGATION BAR */}
      <nav className="top-navbar" aria-label="Main Navigation">
        <div className="nav-logo">
          {/* Put your logo image or text here */}
          <img src={logo} alt="OmniKey Logo" className="logo" />
        </div>

        <div className="nav-user-menu-container">
          <button 
            className="user-menu-toggle"
            onClick={() => setIsUserMenuOpen(!isUserMenuOpen)}
            aria-expanded={isUserMenuOpen}
            aria-haspopup="true"
          >
            <span>Prof. {loggedInUser?.user_name}</span>
            <span className="dropdown-arrow">&#9662;</span>
          </button>

          {isUserMenuOpen && (
            <div className="user-dropdown-menu">
              <button 
                onClick={() => { 
                  setIsUserMenuOpen(false);
                  setLoggedInUser(null); 
                  setCurrentView('login'); 
                }} 
                className="dropdown-signout-btn"
              >
                Sign Out
              </button>
            </div>
          )}
        </div>
      </nav>

      <main className="app-container dashboard-container">
        {/* Live Accessibility Notification Banner */}
        {notification.message && (
          <div className={`notification-banner ${notification.type}`} aria-live="polite">
            {notification.message}
          </div>
        )}

        <header className="dashboard-header">
          <div>
            <h1>Academic Records by OmniKey</h1>
            <p>Manage and review student exam records</p>
          </div>
          <div className="header-actions">
            <button onClick={() => setCurrentView('form')} className="btn primary-btn">+ Add Record</button>
          </div>
        </header>

        {/* FILTER & SEARCH CONTROLS */}
        <section className="controls-section" aria-label="Filter and Search Controls">
          <div>
            <label htmlFor="module-filter"><strong>Filter by Module: </strong></label>
            <select 
              id="module-filter"
              value={selectedModule} 
              onChange={(e) => setSelectedModule(e.target.value)}
            >
              <option value="ALL">All Modules</option>
              <option value="CS101 - Algorithms">CS101 - Algorithms</option>
              <option value="AI402 - Neural Networks">AI402 - Neural Networks</option>
              <option value="SE305 - Software Engineering">SE305 - Software Engineering</option>
            </select>
          </div>

          <div>
            <label htmlFor="assessment-filter"><strong>Filter by Assessment: </strong></label>
            <select 
              id="assessment-filter"
              value={selectedAssessment} 
              onChange={(e) => setSelectedAssessment(e.target.value)}
            >
              <option value="ALL">All Assessments</option>
              <option value="Final Exam">Final Exam</option>
              <option value="Midterm">Midterm</option>
              <option value="Coursework">Coursework</option>
            </select>
          </div>

          <div>
            <label htmlFor="student-search"><strong>Search Student: </strong></label>
            <input 
              id="student-search"
              type="text" 
              placeholder="Type name or ID..."
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
            />
          </div>
        </section>

        {/* RECORDS TABLE */}
        <section className="table-wrapper">
          <table className="records-table">
            <thead>
              <tr>
                <th>Student Name</th>
                <th>Student ID</th>
                <th>Module</th>
                <th>Assessment</th>
                <th>Actions</th>
              </tr>
            </thead>
            <tbody>
              {filteredRecords.length > 0 ? (
                filteredRecords.map((record) => (
                  <tr key={record.id}>
                    <td>{record.student_name}</td>
                    <td>{record.student_id}</td>
                    <td>{record.module_code}</td>
                    <td>
  <div className="assessment-grade">
    <div className="assessment-grade-info">
      <span>{record.assessment}</span>
      <span className="assessment-percentage">
        {record.score}%
      </span>
    </div>

    <div
      className="grade-progress-track"
      role="progressbar"
      aria-valuenow={record.score}
      aria-valuemin="0"
      aria-valuemax="100"
      aria-label={`Grade: ${record.score}%`}
    >
      <div
        className="grade-progress-fill"
        style={{ width: `${Math.min(100, Math.max(0, Number(record.score) || 0))}%` }}
      />
    </div>
  </div>
</td>
                    <td>
                      <button 
                        onClick={() => openModal(record)} 
                        className="btn details-btn"
                        aria-label={`View or edit details for ${record.student_name}`}
                      >
                        View / Edit
                      </button>
                    </td>
                  </tr>
                ))
              ) : (
                <tr>
                  <td colSpan="5" className="empty-state">No matching records found.</td>
                </tr>
              )}
            </tbody>
          </table>
        </section>

        {/* MODAL POPUP */}
        {activeModalRecord && (
          <div className="modal-overlay" role="dialog" aria-modal="true" aria-labelledby="modal-title">
            <div className="modal-content">
              <header className="modal-header">
                <h2 id="modal-title">
                  {isEditing ? `Edit Record: #${activeModalRecord.id}` : `Record Details: #${activeModalRecord.id}`}
                </h2>
                <button 
                  onClick={() => { setActiveModalRecord(null); setIsEditing(false); }} 
                  className="close-modal-btn"
                  aria-label="Close modal"
                >
                  &times;
                </button>
              </header>

              {!isEditing ? (
                <div className="modal-body">
                  <p><strong>Student Name:</strong> {activeModalRecord.student_name}</p>
                  <p><strong>Student ID:</strong> {activeModalRecord.student_id}</p>
                  <p><strong>Module Code:</strong> {activeModalRecord.module_code}</p>
                  <p><strong>Assessment:</strong> {activeModalRecord.assessment}</p>
                  <p><strong>Final Score:</strong> <span className="score-badge">{activeModalRecord.score}/100</span></p>
                  <p><strong>Status:</strong> <span className={`status-tag ${activeModalRecord.status.toLowerCase().replace(' ', '-')}`}>{activeModalRecord.status}</span></p>
                  
                  {activeModalRecord.review_reason && (
                    <p className="review-reason-box"><strong>Review / Appeal Reason:</strong> {activeModalRecord.review_reason}</p>
                  )}
                  
                  <p className="audit-trail"><small>Last modified by: {activeModalRecord.last_modifiedby || 'System'} on {activeModalRecord.last_modified}</small></p>
                </div>
              ) : (
                <form onSubmit={handleUpdateRecord} className="modal-body edit-form">
                  <div className="form-group">
                    <label htmlFor="edit_name">Student Name</label>
                    <input 
                      id="edit_name"
                      type="text" 
                      value={editFormData.student_name || ''} 
                      onChange={(e) => setEditFormData({...editFormData, student_name: e.target.value})} 
                      required 
                    />
                  </div>

                  <div className="form-group">
                    <label htmlFor="edit_score">Score (0-100)</label>
                    <input 
                      id="edit_score"
                      type="number" 
                      min="0" 
                      max="100"
                      value={editFormData.score || ''} 
                      onChange={(e) => setEditFormData({...editFormData, score: e.target.value})} 
                      required 
                    />
                  </div>

                  <div className="form-group">
                    <label htmlFor="edit_status">Status</label>
                    <select 
                      id="edit_status"
                      value={editFormData.status || 'Approved'} 
                      onChange={(e) => setEditFormData({...editFormData, status: e.target.value})}
                    >
                      <option value="Approved">Approved</option>
                      <option value="Pending Review">Pending Review</option>
                      <option value="Grade Appeal">Grade Appeal</option>
                    </select>
                  </div>

                  {(editFormData.status === 'Pending Review' || editFormData.status === 'Grade Appeal') && (
                    <div className="form-group">
                      <label htmlFor="edit_reason">Review / Appeal Reason</label>
                      <input 
                        id="edit_reason"
                        type="text" 
                        value={editFormData.review_reason || ''} 
                        onChange={(e) => setEditFormData({...editFormData, review_reason: e.target.value})} 
                        required 
                      />
                    </div>
                  )}

                  <div className="modal-edit-actions">
                    <button type="submit" className="btn primary-btn">Save Changes</button>
                    <button type="button" onClick={() => setIsEditing(false)} className="btn secondary-btn">Cancel</button>
                  </div>
                </form>
              )}

              <footer className="modal-footer">
                {!isEditing && (
                  <>
                    <button onClick={() => setIsEditing(true)} className="btn edit-btn">Edit Record</button>
                    <button onClick={() => handleDelete(activeModalRecord.id)} className="btn delete-btn">Delete</button>
                  </>
                )}
                <button onClick={() => { setActiveModalRecord(null); setIsEditing(false); }} className="btn secondary-btn">Close</button>
              </footer>
            </div>
          </div>
        )}
      </main>
    </div>
  );
}
